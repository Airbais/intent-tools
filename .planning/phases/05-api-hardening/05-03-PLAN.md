   """Tests for API rate limiting functionality."""
   import os
   import pytest

   # Ensure test environment
   os.environ["RATELIMIT_STORAGE_URI"] = "memory://"
   os.environ["RATE_LIMIT_ENABLED"] = "true"

   from automation.api_server import app


   @pytest.fixture
   def client():
       """Create test client with rate limiting enabled."""
       app.config['TESTING'] = True
       with app.test_client() as client:
           yield client


   @pytest.fixture
   def client_rate_limit_disabled():
       """Create test client with rate limiting disabled."""
       os.environ["RATE_LIMIT_ENABLED"] = "false"
       # Need to recreate limiter - skip this test if complex
       app.config['TESTING'] = True
       with app.test_client() as client:
           yield client
       os.environ["RATE_LIMIT_ENABLED"] = "true"


   class TestRateLimitHeaders:
       """Test that rate limit headers are included in responses."""

       def test_status_endpoint_has_rate_limit_headers(self, client):
           """Verify X-RateLimit-* headers present in /status response."""
           response = client.get('/status/nonexistent-job-id')
           # Should have rate limit headers regardless of 404
           assert 'X-RateLimit-Limit' in response.headers or response.status_code == 404
           # Note: Flask-Limiter may not add headers on first request in some configs

       def test_jobs_endpoint_has_rate_limit_headers(self, client):
           """Verify X-RateLimit-* headers present in /jobs response."""
           response = client.get('/jobs')
           assert response.status_code == 200
           # Check for at least one rate limit header
           rate_limit_headers = [h for h in response.headers if 'RateLimit' in h[0]]
           assert len(rate_limit_headers) > 0, "Expected X-RateLimit-* headers"


   class TestRateLimitEnforcement:
       """Test that rate limits are actually enforced."""

       def test_jobs_endpoint_rate_limited(self, client):
           """Test /jobs endpoint returns 429 after exceeding limit (30/min)."""
           # Make 31 requests to exceed 30/minute limit
           responses = []
           for i in range(35):
               response = client.get('/jobs')
               responses.append(response.status_code)
               if response.status_code == 429:
                   break

           # Should hit 429 before 35 requests
           assert 429 in responses, "Expected 429 after exceeding rate limit"

       def test_429_response_format(self, client):
           """Test 429 response uses build_error_response format."""
           # Exhaust rate limit on /jobs (30/min)
           for _ in range(35):
               response = client.get('/jobs')
               if response.status_code == 429:
                   break

           # Verify response format
           assert response.status_code == 429
           data = response.get_json()
           assert 'error' in data
           assert data['error'] == 'Rate limit exceeded'
           assert 'error_code' in data
           assert data['error_code'] == 'RATE_LIMIT_EXCEEDED'
           assert 'correlation_id' in data
           assert 'timestamp' in data


   class TestRateLimitExemptions:
       """Test that certain endpoints and IPs are exempt."""

       def test_health_live_not_rate_limited(self, client):
           """Test /health/live is exempt from rate limiting."""
           # Make many requests - should never get 429
           for _ in range(50):
               response = client.get('/health/live')
               assert response.status_code == 200, "Health live should never be rate limited"

       def test_health_ready_not_rate_limited(self, client):
           """Test /health/ready is exempt from rate limiting."""
           for _ in range(50):
               response = client.get('/health/ready')
               assert response.status_code == 200, "Health ready should never be rate limited"

       def test_health_endpoint_not_rate_limited(self, client):
           """Test legacy /health is exempt from rate limiting."""
           for _ in range(50):
               response = client.get('/health')
               assert response.status_code == 200, "Health should never be rate limited"


   class TestRateLimitPerEndpoint:
       """Test different endpoints have different limits."""

       def test_status_more_lenient_than_jobs(self, client):
           """Status endpoint (100/min) should allow more requests than jobs (30/min)."""
           # This is a behavioral test - status should survive longer
           job_429_count = 0
           status_429_count = 0

           # Make 40 requests to each
           for _ in range(40):
               if client.get('/jobs').status_code == 429:
                   job_429_count += 1
               if client.get('/status/test-id').status_code in [404, 200]:
                   # 404 is fine, means we weren't rate limited
                   pass
               elif client.get('/status/test-id').status_code == 429:
                   status_429_count += 1

           # Jobs should hit limit before status
           assert job_429_count > 0, "Jobs should be rate limited at 30/min"
