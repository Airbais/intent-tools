"""
Integration tests for dashboard data loading and schema validation.

Tests verify:
1. Schema validator correctly validates dashboard-data.json files
2. Dashboard data loader can load files from each tool
3. tool_type field is present and valid in tool outputs
"""

import pytest
import json
from pathlib import Path
from datetime import datetime

# Import components under test
from dashboard.schema_validator import DashboardDataValidator, validate_dashboard_data
from dashboard.data_loader import ToolDataLoader


class TestSchemaValidator:
    """Tests for the JSON schema validator"""

    @pytest.fixture
    def validator(self):
        return DashboardDataValidator()

    def test_valid_minimal_data(self, validator):
        """Valid data with required fields passes validation"""
        data = {
            "tool_type": "graspevaluator",
            "timestamp": "2025-01-23T10:00:00Z"
        }
        is_valid, errors = validator.validate(data)
        assert is_valid, f"Valid data should pass: {errors}"

    def test_missing_tool_type_fails(self, validator):
        """Missing tool_type field fails validation"""
        data = {
            "timestamp": "2025-01-23T10:00:00Z",
            "summary": {}
        }
        is_valid, errors = validator.validate(data)
        assert not is_valid
        assert any("tool_type" in e for e in errors)

    def test_invalid_tool_type_fails(self, validator):
        """Invalid tool_type value fails validation"""
        data = {
            "tool_type": "unknown_tool",
            "timestamp": "2025-01-23T10:00:00Z"
        }
        is_valid, errors = validator.validate(data)
        assert not is_valid
        assert any("tool_type" in e for e in errors)

    @pytest.mark.parametrize("tool_type", [
        "intentcrawler",
        "graspevaluator",
        "llmevaluator",
        "geoevaluator",
        "rulesevaluator",
        "llmstxtgenerator"
    ])
    def test_all_valid_tool_types(self, validator, tool_type):
        """All known tool types pass validation"""
        data = {
            "tool_type": tool_type,
            "timestamp": datetime.now().isoformat()
        }
        is_valid, errors = validator.validate(data)
        assert is_valid, f"Tool type {tool_type} should be valid: {errors}"

    def test_additional_properties_allowed(self, validator):
        """Additional tool-specific properties are allowed"""
        data = {
            "tool_type": "intentcrawler",
            "timestamp": "2025-01-23T10:00:00Z",
            "discovered_intents": [],
            "custom_field": "allowed"
        }
        is_valid, errors = validator.validate(data)
        assert is_valid, f"Additional properties should be allowed: {errors}"

class TestDataLoader:
    """Tests for the dashboard data loader"""

    @pytest.fixture
    def loader(self):
        return ToolDataLoader()

    def test_loader_initialization(self, loader):
        """Data loader initializes without error"""
        assert loader is not None
        assert loader.tools_root.exists()

    def test_discover_tools(self, loader):
        """Data loader discovers available tools"""
        tools = loader.discover_tools()
        assert isinstance(tools, list)
        # Should find at least some tools if results exist
        # This test is environment-dependent

    def test_detect_tool_type_explicit(self, loader):
        """Tool type detection uses explicit field when present"""
        data = {"tool_type": "graspevaluator"}
        tool_type = loader._detect_tool_type(data)
        assert tool_type == "graspevaluator"

    def test_detect_tool_type_legacy(self, loader):
        """Tool type detection falls back to legacy 'tool' field"""
        data = {"tool": "rulesevaluator"}
        tool_type = loader._detect_tool_type(data)
        assert tool_type == "rulesevaluator"

    def test_detect_tool_type_heuristic(self, loader):
        """Tool type detection falls back to heuristics for old files"""
        # IntentCrawler heuristic pattern
        data = {"discovered_intents": [], "by_section": {}}
        tool_type = loader._detect_tool_type(data)
        assert tool_type == "intentcrawler"


class TestExistingDataFiles:
    """Tests that verify existing dashboard-data.json files can be loaded.

    These tests only run if result files exist in the expected locations.
    """

    @pytest.fixture
    def loader(self):
        return ToolDataLoader()

    @pytest.fixture
    def validator(self):
        return DashboardDataValidator()

    def _get_latest_result(self, loader, tool_name):
        """Get the latest result file for a tool, or None if not found"""
        runs = loader.get_tool_runs(tool_name)
        if not runs:
            return None
        return loader.load_tool_data(tool_name, runs[0][0])

    @pytest.mark.parametrize("tool_name", [
        "intentcrawler",
        "graspevaluator",
        "llmevaluator",
        "geoevaluator",
        "rulesevaluator",
        "llmstxtgenerator"
    ])
    def test_tool_data_loads(self, loader, tool_name):
        """Each tool's latest data file can be loaded"""
        data = self._get_latest_result(loader, tool_name)
        if data is None:
            pytest.skip(f"No result files found for {tool_name}")
        assert data is not None
        assert "_metadata" in data

    @pytest.mark.parametrize("tool_name", [
        "intentcrawler",
        "graspevaluator",
        "llmevaluator",
        "geoevaluator",
        "rulesevaluator",
        "llmstxtgenerator"
    ])
    def test_tool_type_detected(self, loader, tool_name):
        """Each tool's data is correctly identified"""
        data = self._get_latest_result(loader, tool_name)
        if data is None:
            pytest.skip(f"No result files found for {tool_name}")

        detected_type = data.get("_metadata", {}).get("tool_type")
        assert detected_type == tool_name, f"Expected {tool_name}, got {detected_type}"
