    """

        # Validate it's a known type
        known_types = {'intentcrawler', 'graspevaluator', 'llmevaluator',
                       'geoevaluator', 'rulesevaluator', 'llmstxtgenerator'}
        if tool_type in known_types:
            return tool_type
        else:
            self.logger.warning(f"Unknown tool_type: {tool_type}, falling back to heuristics")

    # Check legacy 'tool' field
    if 'tool' in data:
        tool = data['tool']
        if tool in known_types:

    # Fall back to heuristic detection
