```python
"""
Correlation ID middleware for request tracking.

Generates or extracts correlation IDs for each request and binds them
to structlog's context variables for automatic inclusion in all logs.

Usage:
    from logging.correlation import configure_correlation
    configure_correlation(app)

All subsequent logs within a request will include the correlation_id.
"""
import uuid
import structlog
from flask import Flask, request, g

# Standard header name for correlation ID
CORRELATION_HEADER = "X-Correlation-ID"


def bind_correlation_id():
    """
    Generate or extract correlation ID and bind to structlog context.

    Called at the start of each request via Flask's before_request hook.

    The correlation ID:
    1. Is extracted from X-Correlation-ID header if present (allows chaining)
    2. Or generated as a new UUID v4
    3. Is bound to structlog context for all subsequent logs in this request
    4. Is stored in Flask's g object for response header addition
    """
    # Clear previous request's context variables
    structlog.contextvars.clear_contextvars()

    # Get correlation ID from header or generate new one
    correlation_id = request.headers.get(CORRELATION_HEADER)
    if not correlation_id:
        correlation_id = str(uuid.uuid4())

    # Bind to structlog context - all logs will include this
    structlog.contextvars.bind_contextvars(
        correlation_id=correlation_id,
        http_method=request.method,
        http_path=request.path,
        remote_addr=request.remote_addr,
    )

    # Store in Flask's g object for response header
    g.correlation_id = correlation_id


def add_correlation_header(response):
    """
    Add correlation ID to response headers.

    Called after each request via Flask's after_request hook.
    Allows clients to see which correlation ID was used.
    """
    if hasattr(g, 'correlation_id'):
        response.headers[CORRELATION_HEADER] = g.correlation_id
    return response


def configure_correlation(app: Flask):
    """
    Configure correlation ID middleware for Flask application.

    Registers before_request and after_request hooks to:
    1. Extract or generate correlation ID at request start
    2. Bind correlation ID to structlog context
    3. Add correlation ID to response headers

    Args:
        app: Flask application instance

    Note:
        This should be called AFTER configure_structlog() but BEFORE
        any request logging middleware.
    """
    app.before_request(bind_correlation_id)
    app.after_request(add_correlation_header)


def get_correlation_id() -> str:
    """
    Get the current request's correlation ID.

    Returns:
        Correlation ID string, or empty string if not in request context
    """
    return getattr(g, 'correlation_id', '')
```

```python
"""Tests for correlation ID middleware."""
import pytest
import sys
import json
import structlog

sys.path.insert(0, 'automation')

from api_server import app


@pytest.fixture
def client():
    """Create test client."""
    app.config['TESTING'] = True
    with app.test_client() as client:
        yield client


@pytest.fixture(autouse=True)
def reset_structlog():
    """Reset structlog between tests."""
    structlog.reset_defaults()
    yield
    structlog.reset_defaults()


class TestCorrelationIdMiddleware:
    """Test correlation ID generation and propagation."""

    def test_response_has_correlation_id_header(self, client):
        """Response should include X-Correlation-ID header."""
        response = client.get('/health')
        assert 'X-Correlation-ID' in response.headers
        # Should be a valid UUID format
        correlation_id = response.headers['X-Correlation-ID']
        assert len(correlation_id) == 36  # UUID format: 8-4-4-4-12

    def test_provided_correlation_id_is_preserved(self, client):
        """Provided X-Correlation-ID should be used and returned."""
        test_id = 'test-correlation-123'
        response = client.get(
            '/health',
            headers={'X-Correlation-ID': test_id}
        )
        assert response.headers['X-Correlation-ID'] == test_id

    def test_different_requests_get_different_ids(self, client):
        """Each request without correlation ID gets unique ID."""
        response1 = client.get('/health')
        response2 = client.get('/health')

        id1 = response1.headers['X-Correlation-ID']
        id2 = response2.headers['X-Correlation-ID']

        assert id1 != id2

    def test_correlation_id_on_post_endpoint(self, client):
        """Correlation ID should work on POST endpoints."""
        response = client.post(
            '/intentcrawler/analyze',
            json={'url': 'https://example.com'},
            content_type='application/json'
        )
        assert 'X-Correlation-ID' in response.headers

    def test_correlation_id_on_error_response(self, client):
        """Correlation ID should be present even on error responses."""
        response = client.get('/nonexistent')
        assert response.status_code == 404
        assert 'X-Correlation-ID' in response.headers

    def test_correlation_id_on_validation_error(self, client):
        """Correlation ID should be present on validation errors."""
        response = client.post(
            '/intentcrawler/analyze',
            json={'invalid': 'data'},  # Missing required 'url' field
            content_type='application/json'
        )
        assert 'X-Correlation-ID' in response.headers


class TestCorrelationIdModule:
    """Test correlation module functions."""

    def test_get_correlation_id_outside_request(self):
        """get_correlation_id should return empty string outside request."""
        from logging.correlation import get_correlation_id
        # Outside request context, should return empty string
        assert get_correlation_id() == ''

    def test_correlation_header_constant(self):
        """CORRELATION_HEADER should be X-Correlation-ID."""
        from logging.correlation import CORRELATION_HEADER
        assert CORRELATION_HEADER == 'X-Correlation-ID'
```
