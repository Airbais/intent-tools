```python
"""
Request/response logging middleware.

Logs API requests and responses with timing information.
Automatically redacts sensitive data (passwords, tokens, etc.).

Usage:
    from logging.request_logger import configure_request_logging
    configure_request_logging(app)

All requests will be logged with:
- Request: method, path, endpoint, content_type
- Response: status_code, duration_ms, content_length
"""
import time
import structlog
from flask import Flask, request, g

logger = structlog.get_logger(__name__)

# Fields to redact from request logs (case-insensitive)
SENSITIVE_FIELDS = {
    'password', 'passwd', 'pwd',
    'token', 'access_token', 'refresh_token',
    'api_key', 'apikey', 'api-key',
    'secret', 'secret_key',
    'authorization', 'auth',
    'credential', 'credentials',
    'private_key', 'privatekey',
}


def redact_sensitive(data: dict) -> dict:
    """
    Redact sensitive fields from a dictionary.

    Args:
        data: Dictionary potentially containing sensitive fields

    Returns:
        Dictionary with sensitive values replaced by '[REDACTED]'
    """
    if not isinstance(data, dict):
        return data
    return {
        k: '[REDACTED]' if k.lower() in SENSITIVE_FIELDS else (
            redact_sensitive(v) if isinstance(v, dict) else v
        )
        for k, v in data.items()
    }


def log_request_start():
    """
    Log incoming request details.

    Called at the start of each request via Flask's before_request hook.
    Stores start time for duration calculation.
    """
    g.request_start_time = time.time()

    # Log request start with key details
    # Note: correlation_id is automatically included via structlog context
    logger.info(
        "api_request_started",
        endpoint=request.endpoint,
        content_type=request.content_type,
        content_length=request.content_length,
        query_params=redact_sensitive(dict(request.args)) if request.args else None,
    )


def log_request_end(response):
    """
    Log response details with timing.

    Called after each request via Flask's after_request hook.
    Calculates request duration from stored start time.

    Args:
        response: Flask response object

    Returns:
        Unmodified response object
    """
    duration_ms = 0
    if hasattr(g, 'request_start_time'):
        duration_ms = (time.time() - g.request_start_time) * 1000

    # Log request completion with timing
    logger.info(
        "api_request_completed",
        status_code=response.status_code,
        duration_ms=round(duration_ms, 2),
        content_length=response.content_length,
    )

    return response


def configure_request_logging(app: Flask):
    """
    Configure request/response logging middleware for Flask application.

    Registers before_request and after_request hooks to:
    1. Log request start with method, path, content_type
    2. Log request completion with status_code, duration_ms

    Args:
        app: Flask application instance

    Note:
        This should be called AFTER configure_correlation() so that
        correlation_id is available in the structlog context.
    """
    app.before_request(log_request_start)
    app.after_request(log_request_end)
```

```python
"""Tests for request/response logging middleware."""
import pytest
import sys

sys.path.insert(0, 'automation')

from logging.request_logger import redact_sensitive


class TestRedactSensitive:
    """Test sensitive field redaction."""

    def test_redacts_password_field(self):
        """Password fields should be redacted."""
        data = {'username': 'user', 'password': 'secret123'}
        result = redact_sensitive(data)
        assert result['username'] == 'user'
        assert result['password'] == '[REDACTED]'

    def test_redacts_token_field(self):
        """Token fields should be redacted."""
        data = {'token': 'abc123', 'data': 'value'}
        result = redact_sensitive(data)
        assert result['token'] == '[REDACTED]'
        assert result['data'] == 'value'

    def test_redacts_api_key_field(self):
        """API key fields should be redacted."""
        data = {'api_key': 'key123', 'apikey': 'key456'}
        result = redact_sensitive(data)
        assert result['api_key'] == '[REDACTED]'
        assert result['apikey'] == '[REDACTED]'

    def test_redacts_authorization_field(self):
        """Authorization fields should be redacted."""
        data = {'authorization': 'Bearer xyz'}
        result = redact_sensitive(data)
        assert result['authorization'] == '[REDACTED]'

    def test_case_insensitive_redaction(self):
        """Redaction should be case-insensitive."""
        data = {'PASSWORD': 'secret', 'Token': 'abc'}
        result = redact_sensitive(data)
        assert result['PASSWORD'] == '[REDACTED]'
        assert result['Token'] == '[REDACTED]'

    def test_nested_dict_redaction(self):
        """Nested sensitive fields should be redacted."""
        data = {'user': {'name': 'test', 'password': 'secret'}}
        result = redact_sensitive(data)
        assert result['user']['name'] == 'test'
        assert result['user']['password'] == '[REDACTED]'

    def test_non_dict_passthrough(self):
        """Non-dict values should pass through unchanged."""
        assert redact_sensitive("string") == "string"
        assert redact_sensitive(123) == 123
        assert redact_sensitive(None) is None

    def test_empty_dict(self):
        """Empty dict should return empty dict."""
        assert redact_sensitive({}) == {}

    def test_preserves_non_sensitive_fields(self):
        """Non-sensitive fields should not be modified."""
        data = {'url': 'https://example.com', 'max_pages': 10}
        result = redact_sensitive(data)
        assert result == data


class TestRequestLoggingIntegration:
    """Integration tests for request logging middleware."""

    @pytest.fixture
    def client(self):
        """Create test client."""
        from api_server import app
        app.config['TESTING'] = True
        with app.test_client() as client:
            yield client

    def test_request_logging_does_not_break_health(self, client, capsys):
        """Request logging should not break health endpoint."""
        response = client.get('/health')
        assert response.status_code == 200

    def test_request_logging_does_not_break_analyze(self, client):
        """Request logging should not break analyze endpoint."""
        response = client.post(
            '/intentcrawler/analyze',
            json={'url': 'https://example.com'},
            content_type='application/json'
        )
        # Should get 202 Accepted or validation error, not 500
        assert response.status_code in [202, 400]

    def test_request_logging_on_404(self, client):
        """Request logging should work on 404 responses."""
        response = client.get('/nonexistent')
        assert response.status_code == 404
```
