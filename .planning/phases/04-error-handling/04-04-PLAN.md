```python
"""Tests for structured error responses."""
import pytest
from flask import Flask
from error_handlers import (
    build_error_response,
    handle_airbais_exception,
    register_error_handlers
)
from exceptions import (
    ToolNotFoundError,
    JobNotFoundError,
    ToolExecutionError
)
from utils import utc_now_iso


@pytest.fixture
def app():
    """Create Flask app for testing."""
    app = Flask(__name__)
    app.config['TESTING'] = True
    return app


@pytest.fixture
def app_context(app):
    """Provide Flask app context for testing."""
    with app.app_context():
        with app.test_request_context():
            yield


class TestBuildErrorResponse:
    """Tests for build_error_response helper."""

    def test_includes_required_fields(self, app_context):
        """Response includes error, message, timestamp, correlation_id."""
        response, status = build_error_response(
            error="Test Error",
            message="Something went wrong",
            status_code=500
        )
        data = response.get_json()
        assert data['error'] == "Test Error"
        assert data['message'] == "Something went wrong"
        assert 'timestamp' in data
        assert 'correlation_id' in data
        assert status == 500

    def test_includes_error_code_when_provided(self, app_context):
        """error_code included when explicitly provided."""
        response, _ = build_error_response(
            error="Not Found",
            message="Job not found",
            status_code=404,
            error_code="JOB_NOT_FOUND"
        )
        data = response.get_json()
        assert data['error_code'] == "JOB_NOT_FOUND"

    def test_omits_error_code_when_none(self, app_context):
        """error_code omitted when not provided."""
        response, _ = build_error_response(
            error="Error",
            message="Message"
        )
        data = response.get_json()
        assert 'error_code' not in data or data.get('error_code') is None

    def test_includes_details_when_provided(self, app_context):
        """details dict included when provided."""
        response, _ = build_error_response(
            error="Validation Error",
            message="Invalid input",
            details={'field': 'url', 'issue': 'must be HTTPS'}
        )
        data = response.get_json()
        assert data['details'] == {'field': 'url', 'issue': 'must be HTTPS'}

    def test_timestamp_is_utc(self, app_context):
        """Timestamp includes UTC timezone offset."""
        response, _ = build_error_response("E", "M")
        data = response.get_json()
        assert '+00:00' in data['timestamp'] or data['timestamp'].endswith('Z')

    def test_correlation_id_fallback(self, app_context):
        """correlation_id has fallback when middleware hasn't run."""
        response, _ = build_error_response("E", "M")
        data = response.get_json()
        # Should never be None - should have 'none' fallback or actual ID
        assert data['correlation_id'] is not None


class TestHandleAirbaisException:
    """Tests for custom exception handler."""

    def test_tool_not_found_response(self, app_context):
        """ToolNotFoundError returns 404 with correct error_code."""
        e = ToolNotFoundError("missing-tool")
        response, status = handle_airbais_exception(e)
        data = response.get_json()
        assert status == 404
        assert data['error_code'] == "TOOL_NOT_FOUND"
        assert "missing-tool" in data['message']

    def test_job_not_found_response(self, app_context):
        """JobNotFoundError returns 404 with correct error_code."""
        e = JobNotFoundError("job-123")
        response, status = handle_airbais_exception(e)
        data = response.get_json()
        assert status == 404
        assert data['error_code'] == "JOB_NOT_FOUND"

    def test_tool_execution_error_response(self, app_context):
        """ToolExecutionError returns 500 with correct error_code."""
        e = ToolExecutionError("tool", "crashed")
        response, status = handle_airbais_exception(e)
        data = response.get_json()
        assert status == 500
        assert data['error_code'] == "TOOL_EXECUTION_FAILED"


class TestUtcTimestamp:
    """Tests for UTC timestamp utility."""

    def test_utc_now_iso_format(self):
        """utc_now_iso returns RFC 3339 compliant timestamp."""
        ts = utc_now_iso()
        # Should have date, time, and timezone
        assert 'T' in ts  # ISO 8601 separator
        assert '+00:00' in ts or ts.endswith('Z')  # UTC timezone

    def test_utc_now_iso_parseable(self):
        """Timestamp can be parsed back to datetime."""
        from datetime import datetime
        ts = utc_now_iso()
        # Python 3.11+ can parse +00:00 directly
        parsed = datetime.fromisoformat(ts)
        assert parsed.tzinfo is not None
```
